\begin{abstract}
We study how to predict the movement of a missing person for wilderness
search-and-rescue (SAR) and how to fold the two qualitatively distinct kinds
of field evidence --- \emph{negative} (a region was searched and the person was
not found) and \emph{positive} (expert traces indicating presence with a
nonzero misclassification probability) --- back into the predictive density so
that search-region priority \emph{converges} as evidence accumulates. We model
motion by a learnable, segment-constant, underdamped Langevin
stochastic differential equation (SDE) with an exact Gaussian kernel, trained
on 25.98M real GPS points, and we condition its path space through a Doob
$h$-transform with two dual boundary conditions (Dirichlet $0$ exclusion and
Dirichlet $1$ presence). Soft presence evidence is characterized exactly as a
two-point mixture of the two bridges. This paper makes three contributions,
stated with full honest accounting. \textbf{(i) Base model vs.\ baselines.}
The base SDE does not dominate raw energy score, but it leads on
\emph{precision} (best CEP(R50) $437.9$ across six external baselines) and,
after a recalibration step, repairs the 90\%-highest-density-region coverage on
1--3\,h horizons (HDR90 $0.324/0.364\to 0.904/0.899$, NEX-341 1--3\,h bin). \textbf{(ii) Long-horizon
boundary.} Beyond the real-data limit ($\ge$6\,h covered only by a GeoLife
feasibility supplement, $\ge$12\,h a declared data blank), no long-horizon
empirical accuracy is claimed. \textbf{(iii) Dual-bridge convergence.} The
exclusion bridge finds a synthetic target of prior rank 3 in one step at 2.5\%
searched area, and the presence bridge collapses a ``surprise-tail'' target from
rank 27 to rank 2 with a $4.5\times$ CEP50 search-area reduction; on real legs
the mean target rank improves $23.6\to16.6$. The measure-algebraic core of the
theory (mixture identity, TV-Lipschitz robustness, duality, concentration) is
formalized in Lean4 with zero \texttt{sorry}; PDE / $h$-transform existence is
stated at document level and marked, not faked.
\end{abstract}

\section{Introduction}
\label{sec:intro}

\subsection{Scenario and problem}
\label{sec:scenario}
Wilderness search-and-rescue (SAR): a person is reported missing and rescue teams
allocate effort according to a predictive density over the search region. Teams return
two qualitatively distinct kinds of evidence: \emph{negative} evidence (a region was
searched and the person was \emph{not} found) and \emph{positive} evidence (expert
traces --- campfire, footprints --- that indicate presence but carry a nonzero
misclassification probability). This paper studies how to fold both into a single
predictive density so that \emph{search-region priority converges} as evidence
accumulates. The two mechanisms are duals of one conditioning mechanism on a diffusion
path space (Section~\ref{sec:bridges}): an exclusion bridge (Dirichlet-$0$ boundary) and
a presence bridge (Dirichlet-$1$ boundary), and soft evidence is a convex
reweighting of the two (Section~\ref{sec:pres-bridge}).

\subsection{Data acquisition in SAR}
\label{sec:dataacq}
We follow SAR literature \cite{koester2008,stone2004}. Real continuous GPS tracking is
scarce and short; several probability-of-area (POA) approaches instead rely on
\emph{map-based generated scenarios} --- sampling plausible motion behaviors over a
terrain/road map to produce POA fields. We adopt a \emph{dual-track validation}:
model parameters are learned \emph{only} from real tracking data (a 25.98M-point
GeoLife-class leg), while long-horizon and search-effectiveness experiments are
validated on real data where available and on map-based generated scenarios otherwise,
reported honestly \cite{zheng2010geolife,zheng2015trajectory,osm2026}. Generated data is
never used for training. The concrete data accounting is given in
Section~\ref{sec:exp-data}.

\subsection{Contributions}
\label{sec:contrib}
\begin{itemize}
\item \textbf{Dual-conditioning framework.} A negative exclusion bridge and a positive
presence bridge unified as path-space conditioning of a diffusion, with parameterized
misclassification.
\item \textbf{Soft-evidence theory.} An exact mixture characterization of soft presence
evidence (Theorem~\ref{thm:e5}), a scalar likelihood-ratio robustness theorem with a
TV-Lipschitz bound of $\tfrac12$ (Theorem~\ref{thm:e6}), and an honest account of the
theoretical gaps (Theorem~\ref{thm:e7}).
\item \textbf{Convergent SAR loop.} A framework and a concentration argument
(Theorem~\ref{thm:e8}) that independent evidence items concentrate the posterior on the
true presence pattern, with quantitative synthetic + real evidence.
\item \textbf{Formalized core.} A Lean4 layering: the measure-algebraic core lemmas
(L1--L4, E6, the combinatorial part of E8) are formalized with zero \texttt{sorry}; the
PDE / $h$-transform existence theorems are stated at document level, marked, not faked.
\item \textbf{Honest calibration and horizon accounting.} A recalibration step repairing
90\%-region coverage on 1--3\,h, and explicit boundary statements for $\ge$6\,h / $\ge$12\,h.
\item \textbf{Reproducible pipeline.} A learnable underdamped multi-mode SDE (exact
Gaussian kernel, EM/CRPS estimation) plus six external baselines under one protocol,
with a self-contained reproduction package.
\end{itemize}

\section{Problem Formulation}
\label{sec:problem}
Let $X_t\in\R^2$ be the location under a diffusion
\begin{equation}
  dX_t=b(t,X_t)\,dt+\sigma(t,X_t)\,dW_t,\label{eq:sde}
\end{equation}
with generator $\mathcal{L}f=b\cdot\nabla f+\tfrac12 a:\nabla^2 f$, $a=\sigma\sigma^\top$.
The predictive density solves the forward Fokker--Planck equation $\partial_t
p=\mathcal{L}^*p$. A search outcome is negative (region $\Aregion$ searched over
$[t_0,t_1]$, not found) or positive (expert presence evidence with misclassification
probability). Let $\tau_{\Aregion}:=\inf\{s\ge 0:X_s\in\Aregion\}$.

\subsection{Assumptions}
\begin{assumption}[Uniform ellipticity]\label{ass:a1}
$b,\sigma$ are Lipschitz with linear growth and $a$ is uniformly elliptic on $\Aregion$.
\end{assumption}
\begin{assumption}[Identifiability]\label{ass:a2}
$\Delta t\le 60$\,s for parameter identifiability (verified for I-1).
\end{assumption}
\begin{assumption}[Segment-constant modes]\label{ass:a3}
Within a segment the motion mode is constant.
\end{assumption}
\begin{assumption}[Soft evidence]\label{ass:a4}
Positive evidence is a probabilistic presence constraint with a parameterized
misclassification rate (sensitivity / false-alarm).
\end{assumption}

\subsection{Three bridges on one path space}
\label{sec:threebridge}
\begin{itemize}
\item \textbf{endpoint bridge} (hard endpoint $X_T=y$): $\Aregion$ shrinks to a point and the
window to $\{T\}$; the classical Doob bridge.
\item \textbf{exclusion bridge} (negative information): the event
$\{\tau_{\Aregion}>T\}$ --- searched and not entered.
\item \textbf{presence bridge} (positive information): the event
\begin{equation}
  E:=\{\tau_{\Aregion}\in[t_1,t_2]\}=\{\exists s\in[t_1,t_2]:X_s\in\Aregion\},\label{eq:existevent}
\end{equation}
the person \emph{was present} in $\Aregion$ during $[t_1,t_2]$ --- the most general of the
three, a ``set $\times$ time-window'' hit constraint.
\end{itemize}

\paragraph{Terminology (fixed across the paper).}
We use \emph{exclusion bridge} and \emph{presence bridge} as the two canonical
conditioning terms (we never use ``evidence bridge''); the word \emph{existence} appears
only in the E4 presence/existence family (the implementation class name
\texttt{ExistenceConditioning}, the E4 existence-probability PDE, and the
existence-potential Definition / existence-weighted-density Corollary E4b), where it is
synonymous with \emph{presence}. The
hard-endpoint case is the \emph{endpoint bridge} (the classical Doob bridge); the ablation
arm \texttt{arm22} (a soft-endpoint variant, Section~\ref{sec:exp-c}) is \emph{redundant and
not retained} --- it is not a fourth bridge and is not a recommended selection.

\section{Method}
\label{sec:method}

\subsection{Model selection and justification}
\label{sec:model-sel}
We use the I-1 segment-constant underdamped Langevin SDE with state $z=[X,V]\in\R^2$:
\begin{equation}
  dX=V\,dt,\qquad dV=\big[-\Gamma_k V-\kappa X+a_k X+c_k\big]\,dt+g_k\,dW,\label{eq:i1}
\end{equation}
mode $k\sim\mathrm{Categorical}(\pi)$ (constant within a segment, $K=3$); three drift
terms (friction, potential, affine behavioral); noise on velocity only; exact Gaussian
kernel.

\paragraph{Coordinate convention and the default $\kappa=0$.}
The state $X$ in \eqref{eq:i1} is \emph{not} a global latitude/longitude pair.
In the data pipeline (\texttt{code/pipeline\_r1.py::local\_projection}, shipped in
the reproduction package zip1) every trajectory is projected to a local tangent
plane whose origin is that trajectory's own centroid $(\mathrm{lat}_0,\mathrm{lon}_0)$
(equirectangular projection, units of metres): $X=(x,y)$ is the metric displacement
\emph{relative to the trajectory's own centroid}. Every trajectory therefore has a
\emph{different} origin --- there is no single global origin shared across people,
countries, or terrains. Two consequences follow. \emph{(i) The restoring term
$-\kappa X$ is off in the base configuration:} \texttt{code/segment\_constant.py::SegmentConstantSDE}
defaults to $\kappa=0$ (\texttt{code/config.yaml} \texttt{kappa: 0.0}), so the base
model carries no force pulling motion toward a fixed point --- the drift is
$\kappa$-free and there is \emph{no global attractor}. \emph{(ii) The affine
behavioral term $a_k X + c_k$} is a per-mode learned drift in this \emph{local
centroid frame}; because the frame has already absorbed the global translation,
$a_k X + c_k$ describes a \emph{segment-level behavioral tendency relative to the
trajectory itself} (e.g.\ directionality, or a tendency toward or away from its own
centroid), not motion toward any global fixed location. In particular, $\kappa>0$ is
available as an opt-in harmonic potential but is not exercised in the base, and the
base model is invariant under a global translation of the coordinate origin. These
projection and default-$\kappa$ facts are stated in the shipped sources
(\texttt{code/pipeline\_r1.py}, \texttt{code/segment\_constant.py},
\texttt{code/config.yaml}), so the coordinate convention is verifiable against the
delivered code.

\emph{Why this selection:} (i) OU generalization --- one mode with
$\kappa=a=c=0$ recovers OU/CRW; the no-loss gate \texttt{exact-kernel-vs-OU} (mean error
$3.9\times10^{-14}$) verifies exact reduction. (ii) Necessity of physical structure:
$h$-transform conditioning requires the generator $\mathcal{L}$, absent in pure deep
models. (iii) The selected components are the base/reference configuration in the FULL-001
ablation; no variant significantly beats the base (0 win, Section~\ref{sec:exp-c}).

\subsection{Learning and inference}
\label{sec:learning}
EM (warm-start $k$-means $\to$ per-cluster LS $\to$ discrete-to-continuous inverse) or
CRPS scoring \cite{gneiting2007}; same objective at train and deployment. Transfer via
fine-tuning. Inference: exact-kernel rollout, Strang splitting, Fokker--Planck, or CRN MC,
yielding per-region per-horizon POA maps. Training diagnostics (EM NLL and mode prior
collapse) are shown in Figure~\ref{fig:train-conv}.

\subsection{Conditioning bridges}
\label{sec:bridges}
Both bridges act through a Doob $h$-transform: given a positive $\mathcal{L}$-harmonic
potential $h$, the conditioned drift is $b^{h}=b+a\nabla\log h$. The two bridges differ
only in the boundary conditions that determine $h$ (Dirichlet $0$ vs.\ Dirichlet $1$),
the source of the duality in Section~\ref{sec:duality}.

\subsubsection{Negative conditioning: exclusion bridge}
\begin{definition}[Survival probability]\label{def:surv}
$h_{\mathrm{excl}}(t,x):=P(\tau_{\Aregion}>T\mid X_t=x)$, $x\notin\Aregion$.
\end{definition}

\begin{theorem}[Backward Kolmogorov with Dirichlet $0$; E1]\label{thm:e1}
Under Assumption~\ref{ass:a1}, $h_{\mathrm{excl}}$ is the unique bounded solution of
\begin{equation}
  \partial_t h_{\mathrm{excl}}+\mathcal{L}h_{\mathrm{excl}}=0\ \text{on}\
  [0,T]\times(D\setminus\overline\Aregion),\quad
  h_{\mathrm{excl}}\big|_{[0,T]\times\partial\Aregion}=0,\quad
  h_{\mathrm{excl}}(T,\cdot)=\ind_{D\setminus\Aregion}.
\end{equation}
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{theorem}[Survival-weighted density; E2]\label{thm:e2}
The exclusion-conditioned marginal density is
\begin{equation}
  p_{\Excl}(t,x)=\frac{h_{\mathrm{excl}}(t,x)\,p(t,x)}{P(\tau_{\Aregion}>T)},\qquad
  P(\tau_{\Aregion}>T)=\int h_{\mathrm{excl}}(0,x)\,p(0,x)\,dx.
\end{equation}
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{theorem}[Doob $h$-transform; E3]\label{thm:e3}
The law $\P{\cdot\mid\tau_{\Aregion}>T}$ is the $h$-transform of the prior law by
$h_{\mathrm{excl}}$: $\mathcal{L}^{h}f=\mathcal{L}f+(a\nabla\log h_{\mathrm{excl}})\cdot\nabla f$,
with drift $b^{\mathrm{excl}}=b+a\nabla\log h_{\mathrm{excl}}$.
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{corollary}[Hard exclusion baseline; E0]\label{cor:e0}
$p_{\Excl}(t,x)\propto\ind_{x\notin\Aregion}p(t,x)$ is the zero-noise limiting case, used
only as a control.
\end{corollary}

\subsubsection{Positive conditioning: presence bridge}
\label{sec:pres-bridge}
\begin{definition}[Existence potential]\label{def:exist}
$h_{\mathrm{exist}}(t,x):=P\big(\exists s\in[t,t_2]:X_s\in\Aregion\mid X_t=x\big)$.
\end{definition}

\begin{theorem}[Existence probability PDE; E4]\label{thm:e4}
$h_{\mathrm{exist}}$ is the unique bounded solution on $(t_1,t_2)\times(D\setminus\overline\Aregion)$ of
$\partial_t h_{\mathrm{exist}}+\mathcal{L}h_{\mathrm{exist}}=0$,
$h_{\mathrm{exist}}\big|_{[t_1,t_2]\times\partial\Aregion}=1$ (Dirichlet $1$),
$h_{\mathrm{exist}}(t_2,\cdot)=\ind_{\Aregion}$, and for $t<t_1$,
$h_{\mathrm{exist}}(t,x)=P(\tau_{\Aregion}\in[t_1,t_2]\mid X_t=x)$ satisfies the same PDE with
terminal value $h_{\mathrm{exist}}(t_1,\cdot)$. The prior presence probability is
$\pi:=P(E)=h_{\mathrm{exist}}(t_1,x_0)$.
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{corollary}[Existence-weighted density; E4b]\label{thm:e4b}
$p_{\mathrm{exist}}(t,x)=h_{\mathrm{exist}}(t,x)\,p(t,x)/\pi$.
\end{corollary}

\begin{theorem}[Soft evidence = weighted presence constraint (mixture); E5]\label{thm:e5}
Let the expert report $Y\in\{0,1\}$ with sensitivity $\ell_1:=P(Y{=}1\mid E)=1-\beta$ and
false-alarm rate $\ell_0:=P(Y{=}1\mid E^{c})=\alpha$. Given $Y=1$, the posterior law is
\begin{equation}
  \mathbb{Q}=w\,P(\cdot\mid E)+(1-w)\,P(\cdot\mid E^{c}),\qquad
  w=\frac{\ell_1\pi}{Z},\quad Z=\ell_1\pi+\ell_0(1-\pi),
\end{equation}
where $P(\cdot\mid E)$ is the presence bridge (E4) and $P(\cdot\mid E^{c})$ the absence
bridge ($h=1-h_{\mathrm{exist}}$). Equivalently $w=\rho\pi/(\rho\pi+1-\pi)$ with
$\rho:=\ell_1/\ell_0=(1-\beta)/\alpha$.
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{corollary}[Single $h$-transform valid only before the window / before the hit; E5c]\label{thm:e5c}
Let $h_{\mathrm{soft}}(t,x):=\ell_0+(\ell_1-\ell_0)h_{\mathrm{exist}}(t,x)$. For $t<t_1$ (or
inside the window, before $A$ is hit) the Radon--Nikodym density process is
$M_t=h_{\mathrm{soft}}(t,X_t)/Z$, a $\mathbb{P}$-martingale, and Girsanov gives
$b^{\mathbb{Q}}=b+a\nabla\log h_{\mathrm{soft}}$. Once the window is open and the hit state is
path-dependent, the single-$h$ representation degenerates --- exactly why the mixture form
of Theorem~\ref{thm:e5} is the globally correct description (gap G1, Theorem~\ref{thm:e7}).
\end{corollary}

\begin{theorem}[Misclassification robustness via a scalar likelihood ratio; E6]\label{thm:e6}
The posterior depends on $(\alpha,\beta)$ only through
$\lambda:=\log\rho=\log(1-\beta)-\log\alpha$, with
$w(\lambda)=\pi e^{\lambda}/(\pi e^{\lambda}+1-\pi)$ and
$|dw/d\lambda|=w(1-w)\le\tfrac14$. For two settings $\lambda,\lambda'$,
$\|\mathbb{Q}(\lambda)-\mathbb{Q}(\lambda')\|_{\mathrm{TV}}
\le 2|w(\lambda)-w(\lambda')|\le\tfrac12|\lambda-\lambda'|$.
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{remark}[Consistency limits]
$\rho\to\infty$ recovers the hard presence bridge $P(\cdot\mid E)$; $\rho=1$ leaves
$\mathbb{Q}=\mathbb{P}$ unchanged; $\rho\to0$ recovers $P(\cdot\mid E^{c})$. Soft evidence
interpolates continuously between the prior and the hard bridge; the exclusion bridge is
the $\rho\to0$/Dirichlet-$0$ end of the same family.
\end{remark}

\begin{theorem}[Theoretical gaps, honest statement; E7]\label{thm:e7}
(G1) Soft conditioning is \emph{not} a single Markovian $h$-transform; the single-$h$
drift (E5c) is valid only before the window opens / before the hit; the correct object is
the two-component mixture (E5). (G2) Misclassification robustness reduces to the scalar
$\lambda=\log\rho$ with the TV-Lipschitz bound (E6). (G3) The presence/exclusion duality
(Section~\ref{sec:duality}). (G4) \emph{Non-Markov joint constraints} (``present in $A$ and
later in $B$'', multi-region temporal conjunctions, long-memory / fractal noise) break the
single first-hitting-time structure and require path-space conditioning; no closed form is
given here, stated as open.
\end{theorem}

\subsection{Duality of presence and exclusion bridges}
\label{sec:duality}
\begin{proposition}[Complementarity]\label{prop:dual}
On the full horizon, $h_{\mathrm{exist}}:=\P{\tau_{\Aregion}\le T\mid\cdot}$ and
$h_{\mathrm{excl}}:=\P{\tau_{\Aregion}>T\mid\cdot}$ satisfy
$h_{\mathrm{exist}}+h_{\mathrm{excl}}\equiv 1$, hence
$\nabla\log h_{\mathrm{exist}}=-h_{\mathrm{excl}}(1-h_{\mathrm{excl}})^{-1}\nabla\log
h_{\mathrm{excl}}$ and
$b^{\mathrm{exist}}=b-a\,h_{\mathrm{excl}}(1-h_{\mathrm{excl}})^{-1}\nabla\log
h_{\mathrm{excl}}$. For a general window $[t_1,t_2]$,
$h_{\mathrm{exist}}=\P{\tau_{\Aregion}\le t_2\mid\cdot}-\P{\tau_{\Aregion}\le t_1\mid\cdot}$.
\end{proposition}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\begin{proposition}[Mixture duality = numeric check]\label{prop:mixdual}
$\pi\,P(\cdot\mid E)+(1-\pi)\,P(\cdot\mid E^{c})=\mathbb{P}$: the presence and absence
bridges are a two-point mixture decomposition of the prior, and soft evidence merely
reweights this same pair ($w$ replaces $\pi$). Executable check: with
$p_{\mathrm{excl}}\propto h_{\mathrm{excl}}\,p$ and
$p_{\mathrm{exist}}\propto(1-h_{\mathrm{excl}})\,p$,
$\pi\,p_{\mathrm{exist}}+(1-\pi)\,p_{\mathrm{excl}}=p$ pointwise.
\end{proposition}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}

\subsection{Convergence mainline}
\label{sec:convergence}
Let there be $k$ independent evidence items; item $i$ carries region $A_i$, window
$[t_{i1},t_{i2}]$, presence event $E_i$, weight $w_i=\rho_i\pi_i/(\rho_i\pi_i+1-\pi_i)$,
$\rho_i=\ell_{1,i}/\ell_{0,i}$ (soft) or $\rho_i\to0$ (hard exclusion). Let
$\sigma^*\in\{0,1\}^k$ be the true presence pattern.

\begin{theorem}[Evidence concentration; E8]\label{thm:e8}
The joint posterior after $k$ items is a mixture over patterns $\sigma\in\{0,1\}^k$,
\begin{equation}
  \mathbb{Q}_k=\sum_{\sigma\in\{0,1\}^k}W_k(\sigma)\,
  \mathbb{P}\Big(\cdot\;\Big|\;\bigcap_{i:\sigma_i=1}E_i\cap\bigcap_{i:\sigma_i=0}E_i^{c}\Big),\qquad
  W_k(\sigma)\propto\prod_{i:\sigma_i=1}\ell_{1,i}\pi_i\prod_{i:\sigma_i=0}\ell_{0,i}(1-\pi_i).
\end{equation}
For every $\sigma\ne\sigma^*$, the exact weight ratio is a product over mismatched
coordinates
\begin{equation}
  \frac{W_k(\sigma)}{W_k(\sigma^*)}
  =\prod_{i:\sigma_i\ne\sigma^*_i}\frac{\mathrm{wf}_i(\sigma)}{\mathrm{wf}_i(\sigma^*)},
\end{equation}
where matched coordinates cancel (ratio $1$) and each mismatched factor is
$\ell_{0,i}(1-\pi_i)/(\ell_{1,i}\pi_i)$ if $\sigma^*_i=1,\sigma_i=0$, or
$\ell_{1,i}\pi_i/(\ell_{0,i}(1-\pi_i))$ if $\sigma^*_i=0,\sigma_i=1$
(\lean{weight\_ratio\_eq}, \lean{mismatch\_factor}). Under balanced priors
($\pi_i=\tfrac12$), informative items ($\lambda_i=\log(\ell_{1,i}/\ell_{0,i})>0$), and
monotone (nested) deviations $\sigma\subseteq\sigma^*$, this reduces to
$\prod_{i:\sigma_i\ne\sigma^*_i}e^{-|\lambda_i|}\to 0$ at rate
$\sum_{i:\sigma_i\ne\sigma^*_i}|\lambda_i|$ (\lean{nested\_concentration}); by TV-continuity
of the mixture in its weights (as in E6), the posterior search-region priority converges in
total variation to the true-pattern bridge.
\end{theorem}
\begin{proof}\emph{See Appendix~\ref{app:proofs}.}\end{proof}
\textbf{Correction to v3.2.} The claimed equality
$W_k(\sigma)/W_k(\sigma^*)=\prod e^{-|\lambda_i|}$ holds \emph{only} under the three
conditions of item (iii); in general the exact form is the product of item (ii), whose two
directions carry reciprocal factors $e^{\lambda_i}\pi_i/(1-\pi_i)$ and
$e^{-\lambda_i}(1-\pi_i)/\pi_i$ and admit no single positive decay rate $\sum|\lambda_i|$.

\begin{remark}[Honest boundaries]
(i) The \emph{convergence rate} for a specific search policy is not derived here (open).
(ii) $C^{1,2}$-regularity of $h$ requires smooth $b,\sigma$ and $C^2$ boundary; for learned /
piecewise drifts the $h$-transform is a numerical approximation, stated honestly. (iii) G4
is outside this framework. (iv) Horizons beyond the real-data limit are a data blank +
boundary declaration.
\end{remark}

\subsubsection{Negative information: exclusion bridge (NEX-340)}
\label{sec:exp-b-excl}

\textbf{Implementation.} \ExclName
(\texttt{learnable\_sde/inference/exclusion.py}) implements four mechanisms
on the same path space:
\begin{itemize}
\item \textbf{E1} survival-probability backward equation
$h_{\mathrm{excl}}(t,x)=\P{\tau_{\Aregion}>T\mid X_t=x}$ with Dirichlet $0$ on
$\partial \Aregion$;
\item \textbf{E2} survival-weighted density
$p_{\mathrm{Excl}}(t,x)=h_{\mathrm{excl}}\,p\,/\,\P{\tau_{\Aregion}>T}$;
\item \textbf{E3} Doob $h$-transform with conditioned drift
$b^{\mathrm{excl}}=b+a\nabla\log h_{\mathrm{excl}}$;
\item \textbf{E0} hard-exclusion baseline
$p_{\mathrm{Excl}}\propto\ind_{x\notin \Aregion}\,p$ (terminal truncation, control).
\end{itemize}

\textbf{Verification (synthetic, known true parameters).} A single-mode
scene with known true parameters: the true cell starts at prior rank 3/40
and is \emph{found in 1 exclusion step} with searched area fraction 0.025

step'' tension).} The ``prior rank 3'' is the \emph{analytic} prior rank of
the true cell; the search criterion is the MC empirical survival-weighted
density ($n_{\mathrm{path}}{=}10{,}000$, 40 cells), under which the true
cell is the top-survival cell (its paths survive the exclusion condition),
so it is selected as the highest-priority cell in 1 step at 2.5\% area.  The
same record's \texttt{final\_true\_cell\_rank}=22 and
\texttt{true\_cell\_mass}=0.0 are the \emph{post-search} re-ranking after the
searched (true) cell's mass has been zeroed (the \texttt{pick==true\_cell}
degenerate state), so they carry no information about the pre-search priority
and are \emph{not} a claim of final rank 1.  Figure~\ref{fig:excl-conv} (left)

\section{Discussion and Limitations}
\label{sec:discussion}
We state the honest boundaries of this work rather than overclaiming.

\begin{itemize}
\item \textbf{$\ge$12\,h data blank.} Zero in-distribution real segments at
$\ge$6\,h and $\ge$12\,h in the eval set; $\ge$6\,h rests on a GeoLife
feasibility supplement (52 segments / 24 users) and $\ge$12\,h is a declared
data blank. No long-horizon empirical accuracy is claimed.
\item \textbf{Calibration shortfall before recalibration.} The base is
over-confident on the 1--3\,h bin (HDR90 $0.324/0.364$); recalibration
repairs it ($0.904/0.899$) but the $\ge$3h global scale under-repairs
(0.643/0.786), and the calibrated base is judged \emph{redundant} (not
``win'') on ES against the pre-registered threshold.  (The benchmark-set base
HDR90 is $0.543/0.583$, a separate NEX-337 quantity.)
\item \textbf{E2/E3 under-dispersion sensitivity.} Under the uncalibrated
base, path-level exclusion can treat a ``downstream true position'' as shadow
($\mathrm{area\_to\_found}=1.0$ unconditionally); E0 terminal truncation is
the more stable criterion before recalibration.
\item \textbf{I-1 mode collapse.} EM training shows mode-3 prior $\approx1$
(the I-1 segment-constant model collapses toward a single mode), reported in
Figure~\ref{fig:train-conv}.
\item \textbf{22-arm failure-ledger entries.} arm13 (animal-pretrain) is
blocked; the arm17-terrain sub-item ($\Delta=-36.58$) is a loss; the arm17
weather sub-item (uncond/is\_day/weather) was \emph{run} and judged
\emph{redundant} (not ``cancelled''). These are registered in the RC failure
ledger (NEX-241 convention) under entry anchors \texttt{arm2, arm4, arm5,
arm9, arm14} (loss), \texttt{arm17-terrain} (loss), \texttt{arm22}
(redundant, mechanism not passed), and \texttt{arm13} (blocked).
\item \textbf{Convergence demonstrated on 15 real legs only.} The dual-bridge
convergence evidence is 15 real legs; this is a proof-of-mechanism, not a
large-N field validation.
\item \textbf{Terrain negative result is data/representation-limited, not
``terrain-independent''.} The arm17-terrain loss ($\Delta=-36.58$) holds only for our
implemented feature representation (DEM elevation/slope + land-cover raster +
\texttt{has\_map} gate) on the \texttt{has\_map}=1 subset ($79/300$ eval segments, map
coverage fraction $0.538$), with no linear-entity features (roads/ridgelines/rivers).
It therefore cannot be extrapolated to ``terrain is irrelevant'' and does not overturn
the SAR terrain-preference literature; it is registered in the RC failure ledger
(FL-326-arm17t) as a data/representation limitation rather than a model conclusion.
\item \textbf{External reviewer annotation (text pending).} The CEO referenced an
external (\mbox{Stanford} proxy) review on 2026-08-16. Its specific comments are not yet
available to us as text (the attachment is an image and the link requires client-token
interaction); the reviewer text has been requested from the CEO and its items will be
folded into the next revision rather than ignored.
\item \textbf{Theoretical gaps (G4).} Non-Markov joint constraints are open
(Theorem~\ref{thm:e7}); the E8 convergence rate for a specific policy is open.
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}
We present a learnable, segment-constant underdamped Langevin SDE for
lost-person movement prediction, conditioned on both negative and positive
field evidence through a dual Doob $h$-transform (Dirichlet $0$ / Dirichlet
$1$), with soft presence evidence characterized exactly as a two-point mixture
and a scalar TV-Lipschitz robustness bound. Empirically, the base model does
not dominate raw energy score but leads on precision (CEP(R50) $437.9$) and,
after recalibration, repairs 90\%-region coverage on 1--3h; both bridges
converge search-region priority (synthetic exclusion rank $3\to1$ step at
2.5\% area; presence rank $27\to2$ with $4.5\times$ CEP50 reduction; real-leg
mean rank $23.6\to16.6$). The measure-algebraic core is formalized in Lean4
with zero \texttt{sorry}. We do not claim long-horizon accuracy beyond the
real-data limit, and we register the failure paths rather than hiding them.
The framework is a convergent, honest, and reproducible basis for the SAR
prior-optimization stage.

\begin{proof}[Proof of Theorem~\ref{thm:e1} (E1, survival probability)]
\textbf{(i) Martingale property.} By the Markov property (tower rule),
$M_t:=h_{\mathrm{excl}}(t,X_t)=\E{\ind_{\{\tau_{\Aregion}>T\}}\mid\mathcal{F}_t}$ is a
$\mathbb{P}$-martingale on $[0,\tau_{\Aregion}\wedge T)$.
\textbf{(ii) It\^o's formula.} For smooth $h$,
$dh(t,X_t)=(\partial_t h+\mathcal{L}h)(t,X_t)\,dt+(\text{local-martingale})$. As $M_t$ is
a martingale its drift vanishes, hence $(\partial_t+\mathcal{L})h_{\mathrm{excl}}=0$.
\textbf{(iii) Boundary.} For $x\in\partial\Aregion$, path continuity gives
$\tau_{\Aregion}=t$ a.s., so $\ind_{\{\tau_{\Aregion}>T\}}=0$ and $h_{\mathrm{excl}}=0$
(Dirichlet $0$). \textbf{(iv) Terminal.} $h_{\mathrm{excl}}(T,x)=\ind_{D\setminus\Aregion}(x)$.
\textbf{(v) Uniqueness} follows from the parabolic maximum principle.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:e2} (E2, survival-weighted density)]
Bayes: $p_{\Excl}(t,x)=P(\tau_{\Aregion}>T\mid X_t=x)p(t,x)/P(\tau_{\Aregion}>T)
=h_{\mathrm{excl}}(t,x)p(t,x)/P(\tau_{\Aregion}>T)$. The normalizer is the law of total
probability over $X_0$: $P(\tau_{\Aregion}>T)=\int P(\tau_{\Aregion}>T\mid
X_0=x)p(0,x)dx=\int h_{\mathrm{excl}}(0,x)p(0,x)dx$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:e3} (E3, Doob $h$-transform)]
For bounded measurable $f$, define $\mathcal{L}^h f:=h^{-1}\mathcal{L}(hf)$. Second-order
calculus: $\nabla(hf)=h\nabla f+f\nabla h$,
$\nabla^2(hf)=h\nabla^2 f+2(\nabla h)(\nabla f)^\top+f\nabla^2 h$; since $a$ is symmetric,
$a:(\nabla h\otimes\nabla f)=(a\nabla h)\cdot\nabla f$. Hence
$\mathcal{L}(hf)=h\mathcal{L}f+f\mathcal{L}h+(a\nabla h)\cdot\nabla f$. By
Theorem~\ref{thm:e1}, $\mathcal{L}h_{\mathrm{excl}}=0$, so
$h^{-1}\mathcal{L}(hf)=\mathcal{L}f+(a\,h^{-1}\nabla h)\cdot\nabla f
=\mathcal{L}f+(a\nabla\log h)\cdot\nabla f$, and the drift is read off as
$b+a\nabla\log h$. The claim that the conditioned law \emph{is} a Markov diffusion with
this drift requires Girsanov (\emph{document-level}); the generator identity is formal
calculus on $C^2$ test functions.
\end{proof}

\textbf{Corollary~\ref{cor:e0} (E0, hard-exclusion baseline).}
$p_{\Excl}(t,x)\propto\ind_{x\notin\Aregion}p(t,x)$ is the terminal truncation
obtained by setting the survival weight to the indicator (equivalently, letting the noise
vanish so that the path cannot cross $\Aregion$); it is used only as a control, not as the
conditioning mechanism.

\begin{proof}[Proof of Theorem~\ref{thm:e4} (E4, existence-probability PDE)]
Mirror of Theorem~\ref{thm:e1}. The event $\{\exists s\in[t,t_2]:X_s\in\Aregion\}$ has the
Markov projection property; taking the Markov version of $\E{\ind_E\mid\mathcal{F}_t}$ and
applying It\^o's formula plus the martingale property gives
$(\partial_t+\mathcal{L})h_{\mathrm{exist}}=0$. The Dirichlet-$1$ boundary follows from path
continuity and regularity of $\partial\Aregion$: once a path hits $\partial\Aregion$ at
$s\in[t,t_2]$ the event has occurred. The terminal value is $\ind_{\Aregion}$. For $t<t_1$,
the Markov property gives $P(\tau_{\Aregion}\in[t_1,t_2]\mid X_t=x)
=\E{h_{\mathrm{exist}}(t_1,X_{t_1})\mid X_t=x}$, the same PDE.
\end{proof}

\textbf{Corollary~\ref{thm:e4b} (E4b, existence-weighted density).}
Bayes over the presence event $E$, exactly mirroring Theorem~\ref{thm:e2}:
$p_{\mathrm{exist}}(t,x)=P(E\mid X_t=x)\,p(t,x)/P(E)=h_{\mathrm{exist}}(t,x)p(t,x)/\pi$.

\begin{proof}[Proof of Theorem~\ref{thm:e5} (E5, soft-evidence mixture)]
For any event $B$, by Bayes and total probability over $\{E,E^c\}$:
$\mathbb{Q}(B)=P(B\cap\{Y{=}1\})/Z=\big[P(B\cap\{Y{=}1\}\cap E)+P(B\cap\{Y{=}1\}\cap
E^c)\big]/Z$. Given $E$ (resp.\ $E^c$), $\{Y{=}1\}$ is independent of $B$ (the likelihood acts
only through the presence event), so $P(B\cap\{Y{=}1\}\cap E)=\ell_1P(B\cap E)$ and
$P(B\cap\{Y{=}1\}\cap E^c)=\ell_0P(B\cap E^c)$. Thus
$\mathbb{Q}(B)=[\ell_1P(B\cap E)+\ell_0P(B\cap E^c)]/Z$. Normalizing,
$\ell_1P(B\cap E)/Z=(\ell_1\pi/Z)(P(B\cap E)/\pi)=wP(B\mid E)$ and, since
$1-w=\ell_0(1-\pi)/Z$, $\ell_0P(B\cap E^c)/Z=(1-w)P(B\mid E^c)$. The
$\rho$-form is the same expression divided by $\ell_0$. (\emph{Formalized:}
\lean{posterior\_mixture}, \lean{weight\_likelihood\_ratio}.)
\end{proof}

\textbf{Corollary~\ref{thm:e5c} (E5c, single $h$-transform window).}
The Radon--Nikodym density process $M_t=h_{\mathrm{soft}}(t,X_t)/Z$ is a $\mathbb{P}$-martingale
by the same martingale argument as Theorem~\ref{thm:e1}; the Girsanov step that turns it into
the drift $b^{\mathbb Q}=b+a\nabla\log h_{\mathrm{soft}}$ is \emph{document-level} (not
formalized). The degeneration once the window is open is exactly gap G1.

\begin{proof}[Proof of Theorem~\ref{thm:e6} (E6, misclassification robustness)]
The two components $P(\cdot\mid E),P(\cdot\mid E^c)$ do not depend on $(\alpha,\beta)$;
$w$ depends only on $\rho$ (Theorem~\ref{thm:e5}) and $\lambda=\log\rho$. Differentiating,
$dw/d\lambda=w(1-w)$ (\lean{weight\_deriv}); the quadratic bound
$w(1-w)=\tfrac14-(w-\tfrac12)^2\le\tfrac14$ is elementary (\lean{weight\_slope\_bound}).
For any event $B$,
$|\mathbb{Q}(\lambda)(B)-\mathbb{Q}(\lambda')(B)|=|w-w'|\,|P(B\mid E)-P(B\mid E^c)|
\le 2|w-w'|$ since both conditional probabilities lie in $[0,1]$; taking the supremum over
$B$ gives the TV bound $\le 2|w-w'|$ (\lean{mixture\_tv\_bound}). Finally the mean value
theorem with the derivative bound gives $|w(\lambda)-w(\lambda')|\le\tfrac14|\lambda-\lambda'|$
(\lean{e6\_weight\_lipschitz}), and composing yields
$\le 2\cdot\tfrac14|\lambda-\lambda'|=\tfrac12|\lambda-\lambda'|$ (\lean{e6\_tv\_lipschitz}).
\end{proof}

\subsection{Duality}

\begin{proof}[Proof of Proposition~\ref{prop:dual} (complementarity)]
(a) Complementary events: $P(\tau_{\Aregion}\le T\mid\cdot)+P(\tau_{\Aregion}>T\mid\cdot)=1$
(\lean{prob\_compl\_identity}). (b) Since $h_{\mathrm{exist}}=1-h_{\mathrm{excl}}$, the chain rule
gives $\nabla\log h_{\mathrm{exist}}=-\nabla h_{\mathrm{excl}}/(1-h_{\mathrm{excl}})
=-h_{\mathrm{excl}}(1-h_{\mathrm{excl}})^{-1}\nabla\log h_{\mathrm{excl}}$ (using
$\nabla h_{\mathrm{excl}}=h_{\mathrm{excl}}\nabla\log h_{\mathrm{excl}}$), and substitution into
Theorem~\ref{thm:e3} yields the drift relation. (c) The window version is the difference of two
full-horizon survival potentials since
$\{\tau_{\Aregion}\in[t_1,t_2]\}=\{\tau_{\Aregion}\le t_2\}\setminus\{\tau_{\Aregion}\le t_1\}$.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:mixdual} (mixture duality)]
Total probability: $\pi P(\cdot\mid E)+(1-\pi)P(\cdot\mid E^c)=P(\cdot\cap E)+P(\cdot\cap
E^c)=P(\cdot)$. The density form follows from E4b/E2 and Proposition~\ref{prop:dual}:
$\pi p_{\mathrm{exist}}+(1-\pi)p_{\mathrm{excl}}=(h_{\mathrm{exist}}+h_{\mathrm{excl}})p=p$.
This is the L1 mixture identity at $w=\pi$ (\lean{posterior\_mixture}).
\end{proof}

\subsection{Convergence}

\begin{proof}[Proof of Theorem~\ref{thm:e8} (E8, evidence concentration)]
(i) Iterated Bayes on independent evidence multiplies the pattern likelihood by
$\ell_{1,i}$ or $\ell_{0,i}$ and the prior by $\pi_i$ or $1-\pi_i$; the product is the stated
weight. (ii) The weight ratio is a product over mismatched coordinates because matched
factors cancel exactly. (iii) In the nested/balanced/informative case each mismatched factor
is $\ell_{0,i}/\ell_{1,i}=e^{-\lambda_i}=e^{-|\lambda_i|}<1$
(\lean{ratio\_exp\_neg\_logLR}), so the ratio is a product of sub-unit factors concentrating
to $0$. (iv) TV-continuity transfers weight concentration to TV convergence of the marginal
priority. The mixture algebra is formalized; the bridge laws are document-level (E3/E4).
\end{proof}

\subsection{Core lemmas L1--L4 (formalized in Lean4)}

\textbf{L1 (mixture identity).}
$\mathbb Q=w\,\mathbb P(\cdot\mid E)+(1-w)\,\mathbb P(\cdot\mid E^{c})$ with
$w=\ell_1\pi/Z$, $Z=\ell_1\pi+\ell_0(1-\pi)$. \emph{Proof.} This is the statement proved in
the E5 proof above; equivalently, for any event $B$,
$\mathbb Q(B)=P(B\mid Y{=}1)=\left[\ell_1P(B\cap E)+\ell_0P(B\cap E^{c})\right]/Z
=wP(B\mid E)+(1-w)P(B\mid E^{c})$.
(\lean{posterior\_mixture}, \lean{weight\_likelihood\_ratio}.)

\textbf{L2 (TV bound).}
$\|\mathbb Q(w)-\mathbb Q(w')\|_{\mathrm{TV}}\le 2|w-w'|$. \emph{Proof.} For any event $B$,
$|\mathbb Q(w)(B)-\mathbb Q(w')(B)|=|w-w'|\,|P(B\mid E)-P(B\mid E^{c})|\le 2|w-w'|$ since
both conditional probabilities lie in $[0,1]$; the supremum over $B$ is the total-variation
distance. (\lean{mixture\_tv\_bound}.)

\textbf{L3 (logistic slope and limits).}
$w(\lambda)=\pi e^{\lambda}/(\pi e^{\lambda}+1-\pi)$ satisfies $dw/d\lambda=w(1-w)\le1/4$,
$w(0)=\pi$, and is monotone in $\lambda$ with limits $w\to1$ ($\lambda\to\infty$) and
$w\to0$ ($\lambda\to-\infty$). \emph{Proof.} Direct differentiation gives
$dw/d\lambda=w(1-w)$; writing $w(1-w)=\tfrac14-(w-\tfrac12)^2\le\tfrac14$ is elementary, and
monotonicity follows from $w(1-w)\ge0$ on $[0,1]$. (\lean{weight\_deriv},
\lean{weight\_slope\_bound}, \lean{weight\_at\_zero}, \lean{weight\_mono}.)

\textbf{L4 (duality identity).}
On the full horizon, $h_{\mathrm{exist}}+h_{\mathrm{excl}}=1$. \emph{Proof.} The events
$\{\tau_{\Aregion}\le T\}$ and $\{\tau_{\Aregion}>T\}$ are complementary, so their conditional
probabilities sum to $1$. (\lean{prob\_compl\_identity}.)
