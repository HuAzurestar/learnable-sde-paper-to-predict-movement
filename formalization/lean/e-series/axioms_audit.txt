ESeries.posterior_mixture (ℓ₁ ℓ₀ π pBE pBEc : ℝ) (hπ : π ≠ 0) (hπc : 1 - π ≠ 0) (hZ : ℓ₁ * π + ℓ₀ * (1 - π) ≠ 0) :
  (ℓ₁ * pBE + ℓ₀ * pBEc) / (ℓ₁ * π + ℓ₀ * (1 - π)) =
    ℓ₁ * π / (ℓ₁ * π + ℓ₀ * (1 - π)) * (pBE / π) + (1 - ℓ₁ * π / (ℓ₁ * π + ℓ₀ * (1 - π))) * (pBEc / (1 - π))
ESeries.weight_likelihood_ratio (ℓ₁ ℓ₀ π : ℝ) (hℓ₀ : ℓ₀ ≠ 0) (hZ : ℓ₁ * π + ℓ₀ * (1 - π) ≠ 0) :
  ℓ₁ * π / (ℓ₁ * π + ℓ₀ * (1 - π)) = ℓ₁ / ℓ₀ * π / (ℓ₁ / ℓ₀ * π + (1 - π))
ESeries.mixture_tv_bound (a b w w' : ℝ) (ha : 0 ≤ a) (ha1 : a ≤ 1) (hb : 0 ≤ b) (hb1 : b ≤ 1) :
  |w * a + (1 - w) * b - (w' * a + (1 - w') * b)| ≤ 2 * |w - w'|

  |ESeries.weight π lam - ESeries.weight π lam'| ≤ 1 / 4 * |lam - lam'|

  deriv (fun t => ESeries.weight π t) x = ESeries.weight π x * (1 - ESeries.weight π x)
ESeries.mismatch_factor.{u_1} {ι : Type u_1} (ℓ₁ ℓ₀ π : ι → ℝ) {σ σstar : ι → Bool} {i : ι} (hmiss : σ i ≠ σstar i) :
  ESeries.weightFactor ℓ₁ ℓ₀ π σ i / ESeries.weightFactor ℓ₁ ℓ₀ π σstar i =
    if σstar i = true then ℓ₀ i * (1 - π i) / (ℓ₁ i * π i) else ℓ₁ i * π i / (ℓ₀ i * (1 - π i))
ESeries.weight_ratio_eq.{u_1} {ι : Type u_1} [Fintype ι] (ℓ₁ ℓ₀ π : ι → ℝ) (σ σstar : ι → Bool)
